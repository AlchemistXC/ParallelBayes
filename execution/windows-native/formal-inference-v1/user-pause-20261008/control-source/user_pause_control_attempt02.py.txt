def snapshot():
    db = sqlite3.connect('file:'+str(LOCK)+'.formal-journal/tasks.sqlite3?mode=ro', uri=True)
    db.execute('BEGIN')
    counts = {'posterior': {}, 'cache_measurement': {}}
    entries = []
    for key, output, rec, digest, active, p, batch, outcome in db.execute(
            'SELECT key,output,record,checksum,active,protocol,batch,outcome FROM tasks WHERE protocol=?', (PROTOCOL,)):
        entry = json.loads(rec)
        if (fingerprint(entry) != digest or TaskJournal.key(entry['task']) != key or
                TaskJournal._metadata(entry) != (output, active, p, batch, outcome)):
            raise ValueError('Journal identity/checksum differs')
        kind = entry['task']['artifact_kind']
        counts[kind][outcome] = counts[kind].get(outcome, 0) + 1
        if active:
            entries.append(entry)
    db.rollback(); db.close()
    for kind, total in [('posterior',41472), ('cache_measurement',9216)]:
        counts[kind]['not_registered'] = total - sum(counts[kind].values())
    return dict(observed_utc=datetime.now(timezone.utc).isoformat(), counts=counts, active_entries=entries)

class Overlapped(C.Structure):
    _fields_ = [('Internal', C.c_size_t), ('InternalHigh', C.c_size_t),
                ('Offset', W.DWORD), ('OffsetHigh', W.DWORD), ('hEvent', W.HANDLE)]

k = C.WinDLL('kernel32', use_last_error=True)
lock_file = k.LockFileEx
lock_file.argtypes = [W.HANDLE, W.DWORD, W.DWORD, W.DWORD, W.DWORD, C.POINTER(Overlapped)]
lock_file.restype = W.BOOL
unlock_file = k.UnlockFileEx
unlock_file.argtypes = [W.HANDLE, W.DWORD, W.DWORD, W.DWORD, C.POINTER(Overlapped)]
unlock_file.restype = W.BOOL
terminate = k.TerminateProcess
terminate.argtypes = [W.HANDLE, W.UINT]; terminate.restype = W.BOOL

def held_process(pid, creation):
    h = open_process(0x1000 | 0x100000 | 0x0001, False, pid)
    if not h:
        raise C.WinError(C.get_last_error())
    if identity(pid)['creation_filetime'] != creation:
        close(h); raise ValueError('PID reuse; refuse process action')
    return h

def stop_waiter(pid, creation, role):
    import psutil
    h = held_process(pid, creation)
    try:
        command = psutil.Process(pid).cmdline()
        expected = 'publish_formal_study.py' if role=='publisher' else 'finish_formal_study.py'
        if not any(Path(arg).name==expected for arg in command):
            raise ValueError('Different waiter command')
        before = dict(identity=identity(pid), command=command, role=role,
                      reason='User requested pause; cancel future automatic postprocessing/publication')
        atomic_json(OUT/(role+'-stop-intent.json'), before)
        if not terminate(h, 0xE002):
            raise C.WinError(C.get_last_error())
        while True:
            result = wait(h,1000)
            if result==0: break
            if result!=258: raise C.WinError(C.get_last_error())
        atomic_json(OUT/(role+'-stopped.json'), dict(**before, real_process_handle_signaled=True,
                    scientific_task_process_not_targeted=True))
    finally:
        close(h)
